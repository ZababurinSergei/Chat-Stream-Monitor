```typescript
// Тестовый пример, только для повторения структуры
import { TestComponentA } from '..';

export interface EventListener {
  element: EventTarget;
  event: string;
  handler: EventListenerOrEventListenerObject;
}

export const controller = (context: TestComponentA) => {
  let eventListeners: EventListener[] = [];

  const addEventListener = (
          element: EventTarget,
          event: string,
          handler: EventListenerOrEventListenerObject
  ): void => {
    element.addEventListener(event, handler);
    eventListeners.push({ element, event, handler });
  };

  return {
    async init(): Promise<void> {
      // Делегирование кликов через shadowRoot
      addEventListener(context.shadowRoot!, 'click', (e: Event) => {
        const target = e.target as HTMLElement;

        if (target.classList.contains('btn-increment')) {
          (context as any)._actions.handleIncrement();
        }

        if (target.classList.contains('btn-add')) {
          (context as any)._actions.handleAddItem();
        }
      });

      // Обработка формы
      const form = context.shadowRoot!.querySelector('form');
      if (form) {
        addEventListener(form, 'submit', (e: Event) => {
          e.preventDefault();
          const formData = new FormData(form as HTMLFormElement);
          (context as any)._actions.handleFormSubmit(formData);
        });
      }

      // Обработка клавиатуры (пример горячих клавиш)
      addEventListener(context, 'keydown', (e: Event) => {
        const keyboardEvent = e as KeyboardEvent;
        if (keyboardEvent.ctrlKey && keyboardEvent.key === 'i') {
          e.preventDefault();
          (context as any)._actions.handleIncrement();
        }
      });

      // Пример обработки внешних событий
      addEventListener(window, 'component-a:increment', () => {
        (context as any)._actions.handleIncrement();
      });
    },

    async destroy(): Promise<void> {
      // Автоматическая очистка всех слушателей
      eventListeners.forEach(({ element, event, handler }) => {
        element.removeEventListener(event, handler);
      });
      eventListeners = [];
    }
  };
};
```

```typescript
// Тестовый пример, только для повторения структуры
import { TestComponentA } from '..';

/**
 * Actions для Test Component A
 * @module components/test-component-a/actions
 * @version 1.0.0
 * @description Демонстрация обработчиков действий с примерами использования методов BaseComponent
 * @example
 * const actions = await createActions(this);
 * await actions.handleIncrement();
 */
export async function createActions(context: TestComponentA) {
  return {
    handleIncrement: handleIncrement.bind(context),
    handleAddItem: handleAddItem.bind(context),
    handleRemoveItem: handleRemoveItem.bind(context),
    handleFormSubmit: handleFormSubmit.bind(context)
  };
}

/**
 * Увеличивает счетчик с использованием updateElement
 * @async
 * @function handleIncrement
 * @this {TestComponentA} Контекст компонента
 * @returns {Promise<void>}
 * @example
 * await handleIncrement();
 */
async function handleIncrement(this: TestComponentA): Promise<void> {
  (this.state as any).counter++;

  // Использование updateElement для обновления текста
  await this.updateElement({
    selector: '.counter',
    value: `Счетчик: ${(this.state as any).counter}`,
    property: 'textContent'
  });
}

/**
 * Добавляет новый элемент с использованием fullRender
 * @async
 * @function handleAddItem
 * @this {TestComponentA} Контекст компонента
 * @returns {Promise<void>}
 * @example
 * await handleAddItem();
 */
async function handleAddItem(this: TestComponentA): Promise<void> {
  const newItem = `Элемент ${(this.state as any).items.length + 1}`;
  (this.state as any).items.push(newItem);

  // Использование fullRender для полного перерендера
  await this.fullRender(this.state);
}

/**
 * Удаляет элемент по индексу с использованием renderPart
 * @async
 * @function handleRemoveItem
 * @this {TestComponentA} Контекст компонента
 * @param {number} index - Индекс элемента для удаления
 * @returns {Promise<void>}
 * @example
 * await handleRemoveItem(0);
 */
async function handleRemoveItem(this: TestComponentA, index: number): Promise<void> {
  if (index >= 0 && index < (this.state as any).items.length) {
    (this.state as any).items.splice(index, 1);

    // Использование renderPart для частичного рендеринга
    await this.renderPart({
      partName: 'renderList',
      state: { items: (this.state as any).items },
      selector: '.items-list'
    });
  }
}

/**
 * Обрабатывает отправку формы с валидацией
 * @async
 * @function handleFormSubmit
 * @this {TestComponentA} Контекст компонента
 * @param {FormData} formData - Данные формы
 * @returns {Promise<void>}
 * @example
 * const formData = new FormData(form);
 * await handleFormSubmit(formData);
 */
async function handleFormSubmit(this: TestComponentA, formData: FormData): Promise<void> {
  const newItem = formData.get('newItem') as string;

  if (newItem && newItem.trim()) {
    (this.state as any).items.push(newItem.trim());
    await this.fullRender(this.state);

    // Очистка формы через updateElement
    const form = this.shadowRoot!.querySelector('form') as HTMLFormElement;
    if (form) {
      await this.updateElement({
        selector: 'form input[name="newItem"]',
        value: '',
        property: 'value'
      });
    }
  } else {
    // Показ ошибки через модальное окно
    await this.showModal({
      title: 'Ошибка',
      content: 'Пожалуйста, введите значение',
      buttons: [{ text: 'OK', type: 'primary' }]
    });
  }
}
```   

```typescript
// Тестовый пример, только для повторения структуры
export function defaultTemplate({ state = {} }: { state: any; context: any }): string {
  return `
    <div class="test-component-a">
      <h2 class="title">${state.title || 'Test Component A'}</h2>
      <div class="counter">Counter: ${state.counter || 0}</div>
      <button class="btn-increment">Increment</button>
      <button class="btn-add">Add Item</button>
      <div class="items-list">
        ${state.items?.map((item: string, index: number) =>
          `<div class="item">${index + 1}. ${item}</div>`
  ).join('') || 'No items'}
      </div>
    </div>
  `;
}

export function renderList({ state = {} }: { state: any; context: any }): string {
  const items = state.items || [];
  if (!items.length) return '<p>No items</p>';
  return items.map((item: string, index: number) => `<div class="item">${index + 1}. ${item}</div>`).join('');
}
```

```typescript
import { BaseComponent } from '../../base/base-component';
import * as template from './template';
import { controller } from './controller';
import { createActions } from './actions';

export interface TestComponentAState {
  title: string;
  items: string[];
  counter: number;
}

export class TestComponentA extends BaseComponent {
  static override observedAttributes = ['title', 'data-source'];

  constructor() {
    super();
    this._templateMethods = template;
    this.state = {
      title: 'Test Component A',
      items: ['Элемент 1', 'Элемент 2'],
      counter: 0
    };
  }

  override async _componentReady(): Promise<void> {
    this._controller = controller(this);
    this._actions = await createActions(this);
    await this.fullRender(this.state);
  }

  override async _componentAttributeChanged(name: string, oldValue: string | null, newValue: string | null): Promise<void> {
    if (name === 'title' && newValue !== oldValue) {
      this.state.title = newValue || 'Test Component A';
      await this.updateElement({
        selector: '.title',
        value: this.state.title,
        property: 'textContent'
      });
    }
  }

  override async postMessage(event: any): Promise<any> {
    try {
      switch (event.type) {
        case 'ADD_ITEM':
          const newItem = event.data?.item || `Новый элемент ${Date.now()}`;
          (this.state as TestComponentAState).items.push(newItem);
          await this.fullRender(this.state);
          return { success: true, items: (this.state as TestComponentAState).items };

        case 'REMOVE_ITEM':
          const index = event.data?.index;
          if (index >= 0 && index < (this.state as TestComponentAState).items.length) {
            (this.state as TestComponentAState).items.splice(index, 1);
            await this.fullRender(this.state);
            return { success: true, items: (this.state as TestComponentAState).items };
          }
          return { success: false, error: 'Invalid index' };

        case 'INCREMENT_COUNTER':
          (this.state as TestComponentAState).counter++;
          await this.updateElement({
            selector: '.counter',
            value: `Счетчик: ${(this.state as TestComponentAState).counter}`,
            property: 'textContent'
          });
          return { success: true, counter: (this.state as TestComponentAState).counter };

        case 'GET_STATE':
          return { success: true, state: this.state };

        case 'RESET_COMPONENT':
          this.state = {
            title: 'Test Component A',
            items: ['Элемент 1', 'Элемент 2'],
            counter: 0
          };
          await this.fullRender(this.state);
          return { success: true, state: this.state };

        case 'UPDATE_TITLE':
          this.state.title = event.data?.title || 'Test Component A';
          await this.updateElement({
            selector: '.title',
            value: this.state.title,
            property: 'textContent'
          });
          return { success: true, title: this.state.title };

        default:
          console.warn(`[TestComponentA] Неизвестный тип сообщения: ${event.type}`);
          return { success: false, error: 'Unknown message type' };
      }
    } catch (error) {
      this.addError({
        componentName: this.constructor.name,
        source: 'postMessage',
        message: `Ошибка обработки сообщения ${event.type}`,
        details: error
      });
      return { success: false, error: (error as Error).message };
    }
  }

  override async _componentDisconnected(): Promise<void> {
    if (this._controller?.destroy) {
      await this._controller.destroy();
    }
  }
}

if (!customElements.get('test-component-a')) {
  customElements.define('test-component-a', TestComponentA);
}
```

**Правила работы c компонентами:**

* Все созданные методы обязательно асинхронные
* метод fullRender используется только в _componentReady. В других местах запрещено, если есть другие варианты. 
* В коде везде использовать updateElement и если надо renderPart.
* Взаимодействие между компонентами только метод **getComponentAsync**.
* Передача данных между компонентами с помощью метода **postMessage**.
* new CustomEvent и dispatch - запрещено использовать. Используй getComponentAsync и postMessage.
* Использовать только статические ES6 импорты.
* Глобальные стили добавлять в /index.css. В html запрещено добавлять
  Пример:
